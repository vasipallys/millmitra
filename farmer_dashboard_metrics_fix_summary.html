@farmer_bp.route('/analytics/overview', methods=['GET']) @jwt_required() def get_farmer_analytics(): """Get comprehensive farmer analytics overview""" try: # Get basic counts total_farmers = Farmer.query.count() active_farmers = Farmer.query.filter_by(is_active=True).count() # Get contract analytics active_contracts = FarmerContract.query.filter_by(status='active').count() total_contracts = FarmerContract.query.count() # Get procurement analytics from PaddyStock from models.inventory import PaddyStock total_procurement_qty = db.session.query(db.func.sum(PaddyStock.quantity)).scalar() or 0 total_procurement_value = db.session.query( db.func.sum(PaddyStock.quantity * PaddyStock.purchase_price) ).scalar() or 0 # Get payment analytics from FarmerPayment total_payments = db.session.query(db.func.sum(FarmerPayment.amount)).scalar() or 0 # Calculate averages and recent activity avg_land_area = db.session.query(db.func.avg(Farmer.land_area)).scalar() or 0 avg_quality_rating = db.session.query(db.func.avg(Farmer.quality_rating)).scalar() or 0 return jsonify({ 'success': True, 'analytics': { 'total_farmers': total_farmers, 'active_farmers': active_farmers, 'active_contracts': active_contracts, 'total_procurement': total_procurement_qty, 'total_payments': total_payments, 'avg_land_area': round(avg_land_area, 2), 'avg_quality_rating': round(avg_quality_rating, 2) }, 'recent_activity': { ... }, 'ai_analytics': { ... }, 'trend_analysis': { ... } }) except Exception as e: return jsonify({ 'success': False, 'message': f'Error fetching analytics: {str(e)}', 'analytics': { 'total_farmers': 0, 'active_contracts': 0, 'total_procurement': 0, 'total_payments': 0 } }), 500
