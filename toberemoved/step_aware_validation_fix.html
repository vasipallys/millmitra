✅ Step-Aware Validation System

Created validation that only checks fields relevant to the current step

// NEW: Step-aware validation function const validateCurrentStep = (values, step = activeStep) => { validation.clearAll(); switch (step) { case 0: // Basic Information if (!values.name || values.name.trim().length < 2) { validation.addError('Name', 'Name is required and must be at least 2 characters'); } if (!values.phone || !/^[0-9]{10}$/.test(values.phone)) { validation.addError('Phone', 'Phone number must be exactly 10 digits'); } if (!values.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(values.email)) { validation.addError('Email', 'Valid email address is required'); } break; case 1: // Contact Details if (!values.aadhar_number || !/^[0-9]{12}$/.test(values.aadhar_number)) { validation.addError('Aadhar Number', 'Aadhar number must be exactly 12 digits'); } break; case 2: // Farm Details if (!values.total_land_area || values.total_land_area <= 0) { validation.addError('Land Area', 'Land area must be greater than 0'); } break; case 3: // Bank Details if (!values.bank_account_number || values.bank_account_number.length < 8) { validation.addError('Bank Account', 'Bank account number must be at least 8 digits'); } if (!values.bank_ifsc || !/^[A-Z]{4}0[A-Z0-9]{6}$/.test(values.bank_ifsc)) { validation.addError('IFSC Code', 'Invalid IFSC code format'); } break; } return !validation.hasErrors; };
